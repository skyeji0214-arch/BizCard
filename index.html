// Google Apps Script 백엔드 (Code.gs) function doGet() { return HtmlService.createHtmlOutput(getHtmlContent()) .setTitle('명함 등록 관리자') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover'); } function processCardData(data) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName('명함목록'); if (!sheet) { sheet = ss.insertSheet('명함목록'); sheet.appendRow(['등록일시', '이름', '직함/부서', '회사명', '연락처', '이메일', '메모', '이미지링크']); } let fileUrl = ''; if (data.imageBase64) { const folderName = '명함이미지_저장소'; const folders = DriveApp.getFoldersByName(folderName); let folder; if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder(folderName); } const contentType = data.imageBase64.substring(5, data.imageBase64.indexOf(';')); const bytes = Utilities.base64Decode(data.imageBase64.substr(data.imageBase64.indexOf('base64,') + 7)); const blob = Utilities.newBlob(bytes, contentType, 'card_' + new Date().getTime() + '.png'); const file = folder.createFile(blob); fileUrl = file.getUrl(); } sheet.appendRow([ new Date(), data.name || '', data.title || '', data.company || '', data.phone || '', data.email || '', data.memo || '', fileUrl ]); return { status: 'success', message: '성공적으로 저장되었습니다.' }; } catch (error) { return { status: 'error', message: error.toString() }; } } function getHtmlContent() { return `

🎴 명함 간편 등록

📷 명함 촬영 / 이미지 선택
`; }
